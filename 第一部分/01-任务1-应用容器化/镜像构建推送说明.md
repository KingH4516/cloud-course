# 镜像构建与推送命令
## 后端镜像
docker build --no-cache --provenance=false --platform linux/amd64 -t backend:v1 -f backend-Dockerfile .
docker tag backend:v1 swr.cn-north-4.myhuaweicloud.com/scai004712-20231125110/backend:v1
docker push swr.cn-north-4.myhuaweicloud.com/scai004712-20231125110/backend:v1

## 前端镜像
docker build --no-cache --provenance=false --platform linux/amd64 -t frontend:v1 -f frontend-Dockerfile .
docker tag frontend:v1 swr.cn-north-4.myhuaweicloud.com/scai004712-20231125110/frontend:v1
docker push swr.cn-north-4.myhuaweicloud.com/scai004712-20231125110/frontend:v1
